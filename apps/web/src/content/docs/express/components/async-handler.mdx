<Step>Create the Async Handler</Step>

```ts title="src/utils/async-handler.ts"
import { Request, Response, NextFunction } from "express";

export type AsyncRouteHandler = (
  req: Request,
  res: Response,
  next: NextFunction
) => Promise<unknown>;

export function AsyncHandler(fn: AsyncRouteHandler) {
  return function (req: Request, res: Response, next: NextFunction) {
    Promise.resolve()
      .then(() => fn(req, res, next))
      .catch(next);
  };
}
```

<Step>Wrap your route handlers</Step>

import { NextFunction, Request, Response, Router } from "express";
import { AsyncHandler } from "../utils/async-handler";

  "/",
  AsyncHandler(async (req: Request, res: Response, next: NextFunction) => {
    /*
     * business logic

     */
  })
);

export default router;

## Usage Example

Wrap any <Code>async</Code> route handler with <Code>AsyncHandler</Code> so rejected promises are forwarded to your Express error-handling middleware.

```ts
import { AsyncHandler } from "@/utils/async-handler";
```

```ts
import { Request, Response, NextFunction } from "express";
import { AsyncHandler } from "../utils/async-handler";

export const registerUser = AsyncHandler(
  async (req: Request, res: Response, next: NextFunction) => {
    /*
     * business logic
     * your actual code
     */
  }
);
```
