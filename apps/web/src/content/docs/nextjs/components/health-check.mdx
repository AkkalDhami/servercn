
<Step>Add Health Check Endpoint</Step>
```ts title="src/app/api/health/route.ts"
import { NextResponse } from "next/server";

export const dynamic = "force-dynamic";

export async function GET() {
  return NextResponse.json({
    success: true,
    statusCode: 200,
    status: "healthy",
    service: process.env.SERVICE_NAME ?? "nextjs-api",
    version: process.env.npm_package_version ?? "1.0.0",
    message: "Service is up and running",
    timestamp: new Date().toISOString()
  });
}
```

## Usage

Send a <Code>GET</Code> request to the health check endpoint:
<Code>http://localhost:3000/api/health</Code>

**Response:**

```json
{
  "success": true,
  "statusCode": 200,
  "status": "healthy",
  "service": "nextjs-api",
  "version": "0.1.0",
  "message": "Service is up and running",
  "timestamp": "2026-08-06T11:03:02.549Z"
}
```
